  void _updateDigitalSpeed(double target, bool locked) {
    if (!locked) {
      _digitalSpeed = 0;
      _hadFreshFix = false;
      return;
    }
    if (!_hadFreshFix) {
      _digitalSpeed = target.round().clamp(0, 240);
      _hadFreshFix = true;
      return;
    }
    // Small display-only hysteresis prevents 59/60/59 flicker. Because target
    // is frame-interpolated, integer crossings now happen progressively rather
    // than jumping several digits on a GNSS correction frame.
    while (target >= _digitalSpeed + 0.56 && _digitalSpeed < 240) {
      _digitalSpeed++;
    }
    while (target <= _digitalSpeed - 0.56 && _digitalSpeed > 0) {
      _digitalSpeed--;
    }
  }
